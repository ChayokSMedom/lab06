cmake_minimum_required(VERSION 3.10)
project(solver VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(solver solver.cpp)
add_library(solver_lib solver_lib.cpp solver.h)
target_include_directories(solver_lib INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
install(TARGETS solver DESTINATION bin)
install(TARGETS solver_lib DESTINATION lib)
install(FILES solver.h DESTINATION include)
set(CPACK_PACKAGE_NAME "solver")
set(CPACK_PACKAGE_VERSION "1.0.0")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Quadratic equation solver")
set(CPACK_PACKAGE_VENDOR "ChayokSMedom")
set(CPACK_PACKAGE_CONTACT "medvedevachm@gmail.com")
if(WIN32)
    set(CPACK_GENERATOR "ZIP;NSIS")
elseif(APPLE)
    set(CPACK_GENERATOR "DragNDrop")
else()
    set(CPACK_GENERATOR "DEB;RPM")
endif()

include(CPack)
